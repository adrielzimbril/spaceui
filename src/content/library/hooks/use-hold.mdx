## Usage

```tsx
import { useHold } from '@/hooks/use-hold'

export default function SpaceUI({ onDelete }: { onDelete: () => void }) {
  const bind = useHold(() => onDelete(), { delay: 600 })

  return <button {...bind}>Hold to delete</button>
}
```

## API Reference

### Parameters / Props

| Parameter       | Type         | Default    | Description      |
| :-------------- | :----------- | :--------- | :--------------- |
| `callback`      | `() => void` | _required_ | Action callback. |
| `options.delay` | `number`     | `500`      | Hold duration.   |

### Return Value

| Return Type      | Description         |
| :--------------- | :------------------ |
| `HTMLAttributes` | DOM event bindings. |
